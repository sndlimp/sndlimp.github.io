9.3 Correção em lote (replace seguro)

function Substituir-Seguro {
    param($Path, $De, $Para)
    $t = [System.Text.Encoding]::UTF8.GetString([System.IO.File]::ReadAllBytes($Path))
    $n = ([regex]::Matches($t,[regex]::Escape($De))).Count
    if ($n -ne 1) { throw "ocorrencias=$n em $Path" }
    $t = $t.Replace($De,$Para)
    [System.IO.File]::WriteAllText($Path,$t,(New-Object System.Text.UTF8Encoding($false)))
}
