3. Biblioteca comum (ajuste-lib.ps1)

# ajuste-lib.ps1 — funções comuns. Não executar diretamente.
$ErrorActionPreference = 'Stop'
$script:Utf8 = New-Object System.Text.UTF8Encoding($false)
$script:Log  = Join-Path $PSScriptRoot ("log_" + (Get-Date -Format 'yyyyMMdd_HHmmss') + ".txt")

function Write-Log([string]$m) {
    $l = "$(Get-Date -Format 's') $m"
    Write-Host $l
    Add-Content -Path $script:Log -Value $l -Encoding UTF8
}

function Assert-ArvoreLimpa {
    if (git status --porcelain) { throw "Working tree suja. Commite ou faça stash." }
    $b = git branch --show-current
    if ($b -ne 'revisao-manual-v4') { throw "Branch '$b'. Esperado: revisao-manual-v4." }
}

function Invoke-Substituicao {
    param(
        [Parameter(Mandatory)][string]$Arquivo,
        [Parameter(Mandatory)][string]$De,
        [Parameter(Mandatory)][string]$Para,
        [Parameter(Mandatory)][int]$Esperado,
        [Parameter(Mandatory)][string]$Decisao,
        [switch]$Apply
    )
    if ([string]::IsNullOrWhiteSpace($Decisao)) { throw "Sem decisão." }
    if (-not (Test-Path $Arquivo)) { throw "Não existe: $Arquivo" }
    $texto = [IO.File]::ReadAllText((Resolve-Path $Arquivo).Path, $script:Utf8)
    $n = ([regex]::Matches($texto, [regex]::Escape($De))).Count
    if ($n -ne $Esperado) { throw "$Arquivo : esperado $Esperado, encontrado $n para '$De'." }
    Write-Log "[$Decisao] $Arquivo : $n oc. | apply=$($Apply.IsPresent)"
    if ($Apply) {
        [IO.File]::WriteAllText(
            (Resolve-Path $Arquivo).Path,
            $texto.Replace($De, $Para),
            $script:Utf8
        )
    }
}

function Assert-Ausente([string]$padrao, [switch]$Palavra) {
    $files = Get-ChildItem -Recurse -File -Include *.html,*.js,*.xml |
        Where-Object { $_.FullName -notmatch '\\\.git\\|\\_backup' }
    $hits = $files | Select-String -Pattern ([regex]::Escape($padrao)) -SimpleMatch
    if ($hits) { throw "Resíduo '$padrao' em: $($hits | ForEach-Object { $_.Path } | Select-Object -Unique -join ', ')" }
}

function Assert-Presente([string]$padrao) {
    $files = Get-ChildItem -Recurse -File -Include *.html,*.js,*.xml |
        Where-Object { $_.FullName -notmatch '\\\.git\\|\\_backup' }
    $hits = $files | Select-String -Pattern ([regex]::Escape($padrao)) -SimpleMatch
    if (-not $hits) { throw "Ausente e obrigatório: '$padrao'" }
}

function Confirmar-Lote([string]$msg) {
    git add -A
    git commit -m $msg | Out-Null
    Write-Log "Commit: $msg"
}

# ajuste-l8-sitemap.ps1 — lastmod de criação + inclusões/exclusões
param([switch]$Apply)
. "$PSScriptRoot\ajuste-lib.ps1"
Assert-ArvoreLimpa

$backup = 'E:\GitHub_Backup\sndlimp'
if (-not (Test-Path $backup)) { throw "Backup não encontrado: $backup" }

function Get-CreationDate([string]$nome) {
    $src = Join-Path $backup $nome
    if (-not (Test-Path $src)) { return $null }
    return (Get-Item $src).CreationTime.ToString('yyyy-MM-ddTHH:mm:sszzz')
}

if ($Apply) {
    # 8.1 gerar sitemap.xml com lastmod de criação
    $dom = New-Object System.Xml.XmlDocument
    $root = $dom.CreateElement('urlset')
    $root.SetAttribute('xmlns','http://www.sitemaps.org/schemas/sitemap/0.9')

    $urls = @(
      @{u='https://sndlimp.com.br/'; p='index.html'; f='1.0'},
      @{u='https://sndlimp.com.br/economia.html'; p='economia.html'; f='0.9'},
      @{u='https://sndlimp.com.br/servicos.html'; p='servicos.html'; f='0.9'},
      @{u='https://sndlimp.com.br/folder_comercial.html'; p='folder_comercial.html'; f='0.8'},
      @{u='https://sndlimp.com.br/galeria.html'; p='galeria.html'; f='0.8'},
      @{u='https://sndlimp.com.br/faq.html'; p='faq.html'; f='0.7'},
      @{u='https://sndlimp.com.br/novidade.html'; p='novidade.html'; f='0.6'},
      @{u='https://sndlimp.com.br/metodo-sndlimp.html'; p='metodoieee3s.html'; f='0.6'},
      @{u='https://sndlimp.com.br/fontesdevalidacao.html'; p='fontesdevalidacao.html'; f='0.6'},
      @{u='https://sndlimp.com.br/cidades/salvador-do-sul.html'; p='cidades/salvador-do-sul.html'; f='0.8'},
      @{u='https://sndlimp.com.br/cidades/lajeado.html'; p='cidades/lajeado.html'; f='0.8'}
      # + as 21 demais cidades conforme diagnóstico
    )

    foreach ($x in $urls) {
        $d = Get-CreationDate $x.p
        if (-not $d) { Write-Log "SEM DATA: $($x.p) — omitir lastmod"; $d = $null }
        $url = $dom.CreateElement('url')
        $loc = $dom.CreateElement('loc'); $loc.InnerText = $x.u; $url.AppendChild($loc) | Out-Null
        if ($d) {
            $lm = $dom.CreateElement('lastmod'); $lm.InnerText = $d; $url.AppendChild($lm) | Out-Null
        }
        $ch = $dom.CreateElement('changefreq'); $ch.InnerText = 'monthly'; $url.AppendChild($ch) | Out-Null
        $pr = $dom.CreateElement('priority'); $pr.InnerText = $x.f; $url.AppendChild($pr) | Out-Null
        $root.AppendChild($url) | Out-Null
    }

    $dom.AppendChild($root) | Out-Null
    $sw = New-Object System.IO.StringWriter
    $wr = New-Object System.Xml.XmlTextWriter($sw)
    $wr.Formatting = 'Indented'
    $dom.WriteTo($wr)
    $wr.Flush()
    Set-Content 'sitemap.xml' $sw.ToString() -Encoding UTF8

    Assert-Ausente 'salvador_do_sul.html'
    Assert-Ausente 'em_obras.html'

function Buscar([string]$p) {
    $files = Get-ChildItem -Recurse -File -Include *.html,*.js,*.xml |
        Where-Object { $_.FullName -notmatch '\\\.git\\|\\_backup' }
    $files | Select-String -Pattern ([regex]::Escape($p)) -SimpleMatch
}

# verificacao-producao.ps1 — após push e merge em main
param([string]$Base = 'https://sndlimp.com.br')
$ErrorActionPreference = 'Continue'

$xml = [xml](Invoke-WebRequest "$Base/sitemap.xml" -UseBasicParsing).Content
$erros = @()
$ok = 0

foreach ($u in $xml.urlset.url.loc) {
    if ($u -like '*/blog*') { continue }
    try {
        $r = Invoke-WebRequest $u -Method Head -UseBasicParsing -TimeoutSec 15
        if ($r.StatusCode -eq 200) { $ok++ } else { $erros += "$u -> $($r.StatusCode)" }
    } catch { $erros += "$u -> $($_.Exception.Message)" }
}

# URL antiga deve dar 404
try {
    $r = Invoke-WebRequest "$Base/metodoieee3s.html" -Method Head -UseBasicParsing -TimeoutSec 10
    $erros += "metodoieee3s.html ainda responde $($r.StatusCode) — esperado 404"
} catch {
    if ($_.Exception.Response.StatusCode.value__ -eq 404) { Write-Host 'metodoieee3s: 404 OK' }

Riscos e pontos de atenção

4 lotes (1, 2, 3, 7) exigem edição manual — o script só verifica ao final. bloco-ga4-gtm.html (Consent Mode), banner (bloco-cookies.html), menu (<nav> em cada página) e bloco JSON-LD (index.html) não são substituíveis por regex com segurança. Tentativa de automatizar introduz erro estrutural.
Esperado obrigatório é a única defesa contra substituição global catastrófica. Se o CSV mostrar Esperado = 3, e o código tiver 4, o script aborta. Não preencher por estimativa.
folder_comercial.html é exceção com 5551992931816. O lote 4 tem regra explícita para ele. Verificação de resíduo 555196033200 no folder_comercial.html não deve falhar se a intenção for o número oficial ali — o manual permite o segundo —, mas isso é ambiguidade que precisa ser resolvida no CSV.
metodo-sndlimp.html herda o lastmod de metodoieee3s.html (6.3). O script faz isso via $backup. Confirmar que E:\GitHub_Backup\sndlimp\metodoieee3s.html existe.
git mv no lote 6 deve ser executado fora do -Apply global, para evitar commit de arquivo vazio se o title não for encontrado.
lastmod de criação é decisão sua (ADENDO E). O script 8.1 usa Get-Item $src.CreationTime do backup. Se o backup tiver sido copiado após a criação, CreationTime reflete a cópia. Ver §6.5 do manual: cruzar com git log --diff-filter=A.
